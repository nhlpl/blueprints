import math
import numpy as np
import matplotlib.pyplot as plt

def signal_power(m_a_eV):
    # m_a in eV
    # Convert constants
    g = G_AGAMMA * 1e9 # eV^{-1}
    B_T = B
    V_m3 = VOLUME
    Q_val = Q
    rho = RHO_DM # eV/cm^3 -> need to convert to eV/m^3: multiply by 1e6
    rho_m3 = rho * 1e6 # eV/m^3
    # Power in watts (1 eV/s = 1.6e-19 W)
    P_eV_s = (g**2 * rho_m3 * B_T**2 * V_m3 * Q_val) / m_a_eV
    return P_eV_s * 1.6e-19

def noise_power():
    kB = 1.38e-23 # J/K
    f0 = 1.5e9 # Hz (for m=6.18e-6 eV? Actually m=6.18e-6 eV gives f=1.49e9 Hz)

    # For simplicity, use sqrt(t) scaling.
    return (P_sig / P_noise) * math.sqrt(integration_seconds / 1.0) # normalize to 1 sec
